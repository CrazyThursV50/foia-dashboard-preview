function growth(rows, f) {
  const key = d => `${d.level}|${d.Agency}|${d.Key}`
  const by = d3.group(rows.filter(d => f(d) != null), d => d.FY)
  return byYear([...by].filter(([FY]) => by.has(FY - 1)).map(([FY, v]) => {
    const prev = new Map(by.get(FY - 1).map(d => [key(d), f(d)]))
    const both = v.filter(d => prev.has(key(d)))
    const before = d3.sum(both, d => prev.get(key(d)))
    return Object({FY, growth: before ? d3.sum(both, f) / before - 1 : null, n: both.length})
  }))
}

viewof compareWith = {
  // choosing adds to the comparison (no extra button): an agency adds the whole agency; then picking one of its
  // components replaces that whole-agency entry, and further components are added; up to 5; x removes one
  const agencySel = html`<select class="form-select form-select-sm cmp-agency"><option value="">Choose an agency…</option>${agencyNames.map(a => html`<option value="${a}">${a}</option>`)}</select>`
  const compSel = html`<select class="form-select form-select-sm cmp-comp" disabled><option value="">Whole agency (all components)</option></select>`
  const chips = html`<div class="chips"></div>`
  const hint = html`<div class="side-note">Choosing adds it to the comparison; × removes it.</div>`
  const wrap = html`<div class="cmp"><div class="cmp-label">Compare with (up to 5)</div>${agencySel}${compSel}${hint}${chips}</div>`
  wrap.value = []
  let pending = null  // the entry the two lists currently point at
  const changed = () => wrap.dispatchEvent(new Event("input", {bubbles: true}))
  const draw = () => {
    chips.replaceChildren(...wrap.value.map(k => {
      const x = html`<button type="button" title="Remove">×</button>`
      x.onclick = () => { wrap.value = wrap.value.filter(v => v !== k); if (pending === k) pending = null; draw(); changed() }
      return html`<span class="chip">${ENT.get(k).short}${x}</span>`
    }))
    hint.textContent = wrap.value.length >= 5 ? "Up to 5: remove one (×) to add another." : "Choosing adds it to the comparison; × removes it."
  }
  const put = (key, replace) => {
    if (!ENT.has(key)) return
    let v = wrap.value.filter(k => k !== key)
    if (replace && v.includes(replace)) v = v.map(k => k === replace ? key : k)
    else if (v.length < 5) v = [...v, key]
    else return draw()
    wrap.value = [...new Set(v)]; pending = key; draw(); changed()
  }
  agencySel.addEventListener("change", () => {
    const a = agencySel.value
    const comps = ENTITIES.filter(e => e.key.startsWith(`C|${a}|`)).sort((x, y) => d3.ascending(x.short, y.short))
    compSel.replaceChildren(html`<option value="">${comps.length ? "Whole agency (all components)" : "No separate components"}</option>`,
      ...comps.map(e => html`<option value="${e.key}">${e.short}</option>`))
    compSel.disabled = !a || !comps.length
    if (a) put(`A|${a}`, null)
  })
  compSel.addEventListener("change", () => {
    const a = agencySel.value, key = compSel.value || `A|${a}`
    // the first component chosen replaces the whole agency that choosing the agency added
    put(key, pending === `A|${a}` && compSel.value ? pending : null)
  })
  return wrap
}
